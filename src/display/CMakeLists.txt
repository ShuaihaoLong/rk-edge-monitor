set(_display_default OFF)
if(CMAKE_SYSTEM_PROCESSOR STREQUAL "aarch64")
    set(_display_default ON)
endif()
option(RKMON_WITH_LOCAL_DISPLAY "Build LVGL local monitoring display" ${_display_default})
if(NOT RKMON_WITH_LOCAL_DISPLAY)
    return()
endif()
enable_language(C)
set(LVGL_SOURCE_DIR "${PROJECT_SOURCE_DIR}/.local/lvgl/source" CACHE PATH "LVGL v9.4.0 source")
if(NOT EXISTS "${LVGL_SOURCE_DIR}/lvgl.h")
    message(FATAL_ERROR "Run script/prepare-display-deps.sh first (LVGL v9.4.0 required)")
endif()
file(READ "${LVGL_SOURCE_DIR}/lv_version.h" _lv_version)
if(NOT _lv_version MATCHES "LVGL_VERSION_MAJOR 9" OR NOT _lv_version MATCHES "LVGL_VERSION_MINOR 4" OR NOT _lv_version MATCHES "LVGL_VERSION_PATCH 0")
    message(FATAL_ERROR "Local display requires LVGL 9.4.0")
endif()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/lvgl")
find_package(PkgConfig REQUIRED)
pkg_check_modules(LOCAL_DISPLAY REQUIRED IMPORTED_TARGET sdl2 gstreamer-app-1.0 gstreamer-video-1.0 json-c freetype2)
target_link_libraries(lvgl PUBLIC PkgConfig::LOCAL_DISPLAY)
add_executable(rkmon-display main.cpp video_reader.cpp ad_reader.cpp telemetry.cpp)
target_include_directories(rkmon-display PRIVATE "${PROJECT_SOURCE_DIR}/src")
target_compile_options(rkmon-display PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(rkmon-display PRIVATE rkmon::app lvgl::lvgl PkgConfig::LOCAL_DISPLAY)
if(RKMON_BUILD_TESTS)
    add_executable(rkmon_display_data_tests "${PROJECT_SOURCE_DIR}/tests/display_data_tests.cpp" telemetry.cpp)
    target_include_directories(rkmon_display_data_tests PRIVATE "${PROJECT_SOURCE_DIR}/src")
    target_link_libraries(rkmon_display_data_tests PRIVATE rkmon::mqtt PkgConfig::LOCAL_DISPLAY)
endif()
