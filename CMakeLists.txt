cmake_minimum_required(VERSION 3.16)
project(rkmon VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(RKMON_BUILD_TESTS "Build portable middleware tests" ON)
find_package(Threads REQUIRED)
set(SPDLOG_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(SPDLOG_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(SPDLOG_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SPDLOG_FMT_EXTERNAL OFF CACHE BOOL "" FORCE)
set(SPDLOG_FMT_EXTERNAL_HO OFF CACHE BOOL "" FORCE)
set(SPDLOG_USE_STD_FORMAT OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${PROJECT_SOURCE_DIR}/third_party/spdlog/CMakeLists.txt")
    message(FATAL_ERROR "Run git submodule update --init --recursive first")
endif()
add_subdirectory(third_party/spdlog)
add_subdirectory(src/core)
add_subdirectory(src/camera)
add_subdirectory(src/media)
add_subdirectory(src/ai)
add_subdirectory(src/app)
if(RKMON_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
