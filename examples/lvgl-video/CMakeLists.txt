cmake_minimum_required(VERSION 3.16)
project(rkmon_lvgl_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DISPLAY REQUIRED IMPORTED_TARGET sdl2 gstreamer-app-1.0 gstreamer-video-1.0)
if(NOT EXISTS "${LVGL_SOURCE_DIR}/lvgl.h")
    message(FATAL_ERROR "Set LVGL_SOURCE_DIR to LVGL v9.4.0 source")
endif()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" lvgl)
add_executable(lvgl-video-probe main.cpp)
target_link_libraries(lvgl-video-probe PRIVATE lvgl::lvgl PkgConfig::DISPLAY)
target_compile_options(lvgl-video-probe PRIVATE -Wall -Wextra -Wpedantic)
